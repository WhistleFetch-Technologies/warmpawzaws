});

// Get specialists for a specific problem (doctor-level discovery)
app.get("/make-server-3dd53475/customer/specialists/:roleId/:problemId", async (c) => {
  try {
    const roleId = c.req.param('roleId');
    const problemId = c.req.param('problemId');
    const lat = parseFloat(c.req.query('lat') || '0');
    const lng = parseFloat(c.req.query('lng') || '0');
    const radius = parseInt(c.req.query('radius') || '5');
    
    console.log(`👨‍⚕️ Finding specialists for problem: ${problemId}`);
    
    const { findProblemById } = await import('./problem-grid-catalog.tsx');
    const problem = findProblemById(problemId);
    
    if (!problem) {
      return c.json({ error: 'Problem not found' }, 404);
    }
    
    // Get all approved vendors for this role
    const allVendors = await kv.getByPrefix('vendor:') || [];
    const approvedVendors = allVendors.filter((vendor: any) => {
      const vendorRoleId = vendor.roleId || '';
      const roleMatch = vendorRoleId === roleId || 
                        vendorRoleId === roleId.replace('role_', '') ||
                        `role_${vendorRoleId}` === roleId;
      return roleMatch && vendor.onboardingStatus === 'approved' && vendor.isActive !== false;
    });
    
    // Collect all staff from these vendors
    const allSpecialists: any[] = [];
    
    for (const vendor of approvedVendors) {
      const staffMembers = await kv.getByPrefix(`staff:${vendor.vendorId}:`) || [];
      
      for (const staff of staffMembers) {
        // Check if staff specializes in this problem
        const isSpecialist = staff.specializations && 
          Array.isArray(staff.specializations) &&
          staff.specializations.some((spec: string) => problem.mappedSubCategories.includes(spec));
        
        // Include staff if they're a specialist OR if no one has specializations (fallback)
        if (isSpecialist || !staff.specializations) {
          // Get staff's available service styles
          const staffServiceStyles = staff.serviceStyles || vendor.serviceStyles || [];
          
          // Apply location filter if needed
          let includeStaff = true;
          let distance = null;
          
          if (lat !== 0 && lng !== 0 && vendor.location?.coordinates) {
            distance = calculateDistance(
              lat, lng,
              vendor.location.coordinates.lat,
              vendor.location.coordinates.lng
            );
            includeStaff = distance <= radius;
          }
          
          if (includeStaff) {
            allSpecialists.push({
              ...staff,
              vendorId: vendor.vendorId,
              vendorName: vendor.businessName,
              vendorLocation: vendor.location,
              vendorRating: vendor.rating,
              availableServiceStyles: staffServiceStyles,
              distance,
              isVerifiedSpecialist: isSpecialist
            });
          }
        }
      }
    }
    
    // Sort by distance if location provided, otherwise by rating
    if (lat !== 0 && lng !== 0) {
      allSpecialists.sort((a, b) => (a.distance || 0) - (b.distance || 0));
    } else {
      allSpecialists.sort((a, b) => (b.vendorRating || 0) - (a.vendorRating || 0));
    }
    
    console.log(`✅ Found ${allSpecialists.length} specialists`);
    
    return c.json({
      success: true,
      problem,
      specialists: allSpecialists,
      count: allSpecialists.length
    });
  } catch (error) {
    console.error('❌ Error finding specialists:', error);
    return c.json({ error: String(error) }, 500);
  }
});

Deno.serve(app.fetch);
